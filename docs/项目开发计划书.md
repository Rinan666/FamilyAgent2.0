```

  ✅ 题库管理（LLM生成 + 人工审核）
  ✅ 自适应抽题（贝叶斯知识追踪）
  ✅ 智能批改（步骤级评分）

  ❌ 快捷日记（Phase 2）
  ❌ 家族知识库（Phase 2）
  ❌ 镜像Agent（Phase 3）
  ❌ 其他学科（Phase 1.5）
  ❌ 小程序版本（Phase 2）
```

**数据库核心表（Phase 1）**：

```sql
-- 用户表
CREATE TABLE users (
    id BIGSERIAL PRIMARY KEY,
    username VARCHAR(50) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    nickname VARCHAR(100),
    avatar_url VARCHAR(500),
    role VARCHAR(20) DEFAULT 'USER',
    status VARCHAR(20) DEFAULT 'ACTIVE',
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW()
);

-- 家族表
CREATE TABLE families (
    id BIGSERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    description TEXT,
    created_by BIGINT REFERENCES users(id),
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW()
);

-- 家族成员
CREATE TABLE family_members (
    id BIGSERIAL PRIMARY KEY,
    family_id BIGINT REFERENCES families(id),
    user_id BIGINT REFERENCES users(id),
    role VARCHAR(20) DEFAULT 'MEMBER',  -- OWNER, ADMIN, MEMBER, GUEST
    permissions JSONB DEFAULT '{}',
    joined_at TIMESTAMP DEFAULT NOW(),
    UNIQUE(family_id, user_id)
);

-- 知识点树
CREATE TABLE knowledge_points (
    id BIGSERIAL PRIMARY KEY,
```

```python
# ai-service/app/engine/knowledge_tracker.py

class BayesianKnowledgeTracker:
    """
    贝叶斯知识追踪引擎
    
    核心参数（可调）:
    - p_learn: 学习概率 (学生通过一次正确练习掌握知识点的概率)
    - p_guess: 猜测概率 (未掌握但猜对的概率)
    - p_slip: 失误概率 (已掌握但答错的概率)
    - p_forget: 遗忘概率 (已掌握但遗忘的概率)
    """
    
    def __init__(self):
        # 默认参数，后续根据数据拟合
        self.p_learn = 0.15
        self.p_guess = 0.20
        self.p_slip = 0.10
        self.p_forget = 0.03
    
    def update(self, prior_mastery: float, is_correct: bool) -> float:
        """
        根据答题结果更新掌握概率
        
        P(mastered|correct) = P(mastered) * (1-P(slip)) 
                              / (P(mastered)*(1-P(slip)) + (1-P(mastered))*P(guess))
        """
        if is_correct:
            numerator = prior_mastery * (1 - self.p_slip)
            denominator = numerator + (1 - prior_mastery) * self.p_guess
        else:
            numerator = prior_mastery * self.p_slip
            denominator = numerator + (1 - prior_mastery) * (1 - self.p_guess)
        
        posterior = numerator / denominator if denominator > 0 else prior_mastery
        
        # 应用遗忘
        posterior *= (1 - self.p_forget)
        
        return min(0.99, max(0.01, posterior))
    
    def get_zone_of_proximal_development(self, profiles: dict) -> list:
        """返回最近发展区的知识点（掌握概率在0.3-0.7之间）"""
        return [kp for kp, p in profiles.items() if 0.3 <= p <= 0.7]
    
    def adaptive_select(self, questions: list, profiles: dict, n: int = 5) -> list:
        """自适应抽题：优先最近发展区，兼顾复习"""
        zpd = self.get_zone_of_proximal_development(profiles)
        
        # 70% 最近发展区 + 30% 已掌握(防遗忘)
        zpd_questions = [q for q in questions if q.kp_id in zpd]
        review_questions = [q for q in questions if q.kp_id not in zpd]
        
        n_zpd = min(int(n * 0.7), len(zpd_questions))
        n_review = n - n_zpd
        
        selected = (
            random.sample(zpd_questions, n_zpd) +
            random.sample(review_questions, n_review)
        )
        
        return selected
```

### 6.2 LLM输出质量保障

这是本项目最特殊的质量挑战。AI讲题的输出不能像传统软件一样精确断言，需要特殊策略：

```python
# ai-service/tests/test_tutor_agent.py

class TestTutorQuality:
    """讲题质量评估"""
    
    # 规则检查（确定性）
    def test_no_direct_answer(self):
        """验证Agent不会直接给出答案"""
        response = tutor_agent.explain(question=...)
        assert answer_value not in response  # 答案数字不出现在讲解中
    
    def test_socratic_questions_present(self):
        """验证包含引导式提问"""
        response = tutor_agent.explain(question=...)
        questions = extract_questions(response)
        assert len(questions) >= 2  # 至少2个引导性问题
    
    def test_step_by_step_structure(self):
        """验证分步骤讲解"""
        response = tutor_agent.explain(question=...)
        assert has_step_markers(response)  # 有"第一步""第二步"等标记
    
    # 模型评判（使用另一个模型评估）
    def test_explanation_quality_score(self):
        """用评判模型打分"""
        response = tutor_agent.explain(question=...)
        score = evaluator_llm.judge(
            rubric=["是否启发思考", "是否循序渐进", "是否通俗易懂"],
            explanation=response,
            ground_truth=step_by_step_solution
        )
        assert score >= 7.0 / 10.0
    
    # 数学正确性（确定性验证）
    def test_math_steps_correct(self):
        """用sympy验证数学步骤"""
        response = tutor_agent.explain(question=...)
        math_steps = extract_math_expressions(response)
        for step in math_steps:
            assert sympy_verify(step, expected_result)
```
